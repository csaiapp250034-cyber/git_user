-- ========================================================
-- SQL Practice: Creating relational database tables
-- Focus: Primary keys, constraints, and data types
-- ========================================================

CREATE TABLE Companies (
    Company_ID    INT PRIMARY KEY,
    Company_Name  VARCHAR(100) NOT NULL,
    CEO           VARCHAR(100),
    Industry      VARCHAR(100)
);
-- ========================================================
-- Practice Exercise: Ingesting sample records into Companies
-- Practicing data types, multi-row insertion & syntax
-- ========================================================

INSERT INTO Companies (Company_ID, Company_Name, CEO, Industry)
VALUES
    (101, 'Microsoft',  'Satya Nadella',    'Technology & Cloud'),
    (102, 'Google',     'Sundar Pichai',    'Technology & AI'),

    (104, 'Adobe',      'Shantanu Narayen', 'Creative Software'),
    (105, 'Salesforce', 'Marc Benioff',     'Enterprise CRM & Cloud');
-- ========================================================
-- Practice Query: Retrieving formatted table dataset
-- ========================================================

SELECT Company_ID, Company_Name, CEO, Industry 
FROM Companies 
ORDER BY Company_ID ASC;
